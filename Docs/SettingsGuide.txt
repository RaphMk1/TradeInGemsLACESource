[Settings]
; This is the documentation guide for the settings available in Trade in Gems - Loot As Currency Edition (LACE) v3.0

; it is recommended to save your game before editing any settings in the .ini file located in SKSE\Plugins\TradeInGemsLACE\TradeInGemsLACE.ini
; the settings file can be refreshed by game boot, game load, new games, mod menu and console command.

; =============================
; [HOTKEYS & MOD STARTUP STATE]
; =============================
; bEnabled defines if the mod's trading mode will be on or off when you boot the game or a new game is started, it isn't an off switch for the Mod! 
; to completely disable the mod, disable it in your mod loader, or move the mod folder elsewhere. even though it isn't a true mod off switch,
; disabling trading mode reduces most of the calculations done by this mod to the bare minimum and reveals trade items for looting and buying.
bEnabled = true

; DirectX Scan Codes for 2 hotkeys the mod has, with some suggestions
; 	13  = Equals (=) [Current Default for uToggleKey]
; 	26  = Left Bracket ([)
; 	27  = Right Bracket (])
; 	41  = Tilde (~)
; 	55  = Numpad Asterisk (*)
; 	34  = G [Current Default for uWealthKey]
; 	21  = Y 
; 	46  = C
;	181 = / (numpad)
;	55 = * (numpad)
; What they do:
; 	uToggleKey controls the item trading mode activation during game, being intended to use multiple times during play to switch between item trading and vanilla behavior.
; 	uWealthKey brings some information about variables of interest the mod is dealing with, displaying it in a way you can adjust with other settings.
; 	it also refreshes the mod data about the game values for fBarterMin and fBarterMax (core barter price variables), this is for support in case any mod edits these during play.
;	currently it also reloads the settings file, but eventually this feature will be moved to a console command.
uToggleKey = 13
uWealthKey = 34

; =============================
; [ECONOMY & GAMEPLAY SETTINGS]
; =============================
; Item worth calculation modes
; 0 = Currency Mode (Buy Prices)
; 1 = Barter Mode (Sell Prices with modifier)
; 2 = Fixed Mode (Base Prices) 
; Currency Mode tries to make items have the same value as the one needed to buy them from the NPC you are trading with. Its worth calculation Min is 105% of Base, just like buy prices.
; Barter Mode is vanilla-like sell prices with a custom modifier setting. Its worth calculation Max is 100% of Base, just like sell prices are in vanilla.
; Fixed Mode is for keeping the currency items valued at 100% of their current base prices, unchanged by modifiers that don't affect base prices, and is the fastest mode for calculations.
iItemValueMode = 0

; Sorting priority modes for spending trade items during purchases:
; 0 = Worst Value/Weight First (Spends worst V/W ratio items first) - Default
; 1 = Best Value/Weight First (Spends best V/W ratio items first)
; 2 = Heaviest First (Spends heaviest items for trading first)
; 3 = Lightest First (Spends lightest items for trading first)
; 4 = Cheapest First (Spends cheaper items first)
; 5 = Most Expensive First (Spends more valuable items first)
; 6 = No Sorting (Preserves inventory load order sequence)
iSpendingOrderMode = 0

; setting that multiplies the calculated value of an item in Barter Mode.
fBarterModeMultiplier = 1.0

; Loot-First spending priority setting to support those who play with default 0 weight gold or want to try custom lists that include items with worse value/weight than gold.
; true  = transactions will automatically deduct items from your patched items first, real gold coins are only spent as a last resort if items run dry.
; false = default gold-first spending priority mode, item trading only happens when you run out of real gold coins.
bSpendItemsBeforeGold = true

; challenge mode for the LACE framework, making coins bound by an ancient, flesh-consuming curse made against the Dragonborn, that triggers when they are used in dialogue or Barter. 
; this should be paired with bSpendItemsBeforeGold = true to use items as a safety barrier against real gold coin spending, unless you want a true cursed experience.
; warning: disabling trading mode won't help against the curse. Difficulty can be increased with Barter Mode and a multiplier below 1.0.
; true  = giving gold coins drains a portion of your character's life force, causing health damage
; false = disabled, spending gold coins does not inflict health damage to your character (Default)
bCursedGoldMode = false

; Setting that includes character's Speechcraft power modifiers (potions, enchants, blessings).
; *due to how currency mode works, activating this in it will reduce item worth with higher modifiers, this can help making the mode more balanced if you also buy them a lot.
bIncludePowerModifiers = true

; Setting that includes Speechcraft skill perks (Haggling, Allure, etc) into item valuation. Also supports many perk overhauls and economy affecting mods that use Perk Entries.
bIncludePerkModifiers = true

; Prevents ghost gold injection (when items are hidden and you get gold for trading) when interacting with active followers.
bBlockFollowerInjection = true

; Injects ghost gold the moment you interact with an NPC, right before dialogue. Ensures dialogue checks such as hiring mercenaries evaluate your full converted traded item wealth
; immediately on the first talk. This setting works by activating a parallel event handler that triggers when you press the activate hotkey (the one used to talk to npcs and pick
; items or open chests) to the main slower one that activates during dialogue. Deactivating it will reduce this mod's footprint from that hotkey's usage, but it will make NPCs not always 
; know when you have enough currency on first interaction dialogues, plus it reduces the chances an injection attempt will be successful, as the number of attempts will reduce.
bEarlyGoldInjection = true

; Form ID of the base gold currency item in Hex format. Default is 0x0000000F (Vanilla Septim). Shouldn't be modified, but it's there for compatibility and debug.
sGoldFormID = 0x0000000F

; ========================

sCurrencyName = Loot

; iItemNotificationMode is used to configure the display mode of traded item notifications.
; 0 = Fires a notification for every single distinct traded item, expect notifications for a long time when bulk trading
; 1 = Lists just the first few items in a singular notification, font can look small if item names are big

iItemNotificationMode = 0

; Wealth Hotkey Visibility Options (true or 1 = Show row, false or 0 = Hide row)
